cmake_minimum_required(VERSION 3.16)
project(MiniAccel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MINIACCEL_SANITIZER "" CACHE STRING "Sanitizer: address-undefined or thread")
if(MINIACCEL_SANITIZER STREQUAL "address-undefined")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(MINIACCEL_SANITIZER STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
elseif(NOT MINIACCEL_SANITIZER STREQUAL "")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
find_package(Threads REQUIRED)
add_library(miniaccel src/core/task_state.cpp src/core/thread_pool.cpp src/backends/cpu/cpu_backend.cpp)
target_include_directories(miniaccel PUBLIC include PRIVATE src)
target_link_libraries(miniaccel PUBLIC Threads::Threads)
target_compile_options(miniaccel PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
foreach(test status task_state bounded_queue thread_pool cpu_backend)
  if(test STREQUAL "cpu_backend")
    set(test_dir integration)
  else()
    set(test_dir unit)
  endif()
  add_executable(test_${test} tests/${test_dir}/test_${test}.cpp)
  target_include_directories(test_${test} PRIVATE src tests)
  target_compile_options(test_${test} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(test_${test} PRIVATE miniaccel)
  if(NOT test STREQUAL "cpu_backend")
    add_test(NAME ${test} COMMAND test_${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 60)
  endif()
endforeach()

foreach(id 006 007 008 009 010)
  add_test(NAME UT-${id} COMMAND test_cpu_backend ${id})
  set_tests_properties(UT-${id} PROPERTIES TIMEOUT 60)
endforeach()
